class Group {
  constructor (start, end, size) {
    this.start = start;
    this.end = end;
    this.size = size;
  }
}

/**
 * Decodes the 10-bit sector code.
 * @param {number} sectorCode raw sector code (0-1023)
 * @returns {number|Group|string} the sector number, group of sectors, or 'reserved' if reserved byte
 */
function decodeGriesserSector (sectorCode) {
  if (sectorCode === 0) return 'reserved';

  // Check if the least significant bit is 1 (individual sector)
  if ((sectorCode & 0x01) === 1) {
    return (sectorCode >> 1) + 1;
  }

  // Group of sectors
  const groupSize = sectorCode & -sectorCode; // Lowest set bit value

  // Calculate shift based on the position of the lowest set bit
  const shift = groupSize.toString(2).length;
  const baseSector = (sectorCode >> shift) * groupSize;

  return new Group(baseSector, baseSector + groupSize, groupSize);
}

/**
 * Returns the German command name for a command code.
 * @param {number} command
 * @returns {string}
 */
function griesserCommand (command) {
  switch (command) {
    case 1: return 'Fahrbefehl';
    case 2: return 'Wertkorrektur';
    case 3: return 'Automatikzustand';
    case 4: return 'Sperre setzen/löschen';
    case 5: return 'Bedienungs Code';
    case 6: return 'Szene setzen';
    case 7: return 'Sonderbefehl';
    case 8: return 'Datum';
    case 9: return 'Zeit synchronisieren';
    case 10: return 'Sensorwertmitteilung';
    case 11: return 'Busüberwachung';
    case 16: return 'Fahrbereichsgrenzen für Sicherheitsfahrbefehle';
    case 17: return 'Fahrbereichsgrenzen für Sicherheitsfahrbefehle';
    case 19: return 'Fahrbereichsgrenzen für Sicherheitstasterbefehle';
    case 20: return 'Fahrbereichsgrenzen für Sicherheitstasterbefehle';
    case 22: return 'Fahrbereichsgrenzen für Automatiktasterbefehle';
    case 23: return 'Fahrbereichsgrenzen für Automatiktasterbefehle';
    case 24: return 'Fahrbereichsgrenzen für Automatiktasterbefehle';
    default: return 'unbekannter Befehl: ' + command;
  }
}

/**
 * Decodes the command byte.
 * @param {number} commandCode raw command code (0-63)
 * @returns {{ raw: number, type: string }} raw value and human-readable type
 */
function decodeGriesserCommand (commandCode) {
  const type = griesserCommand(commandCode);
  return {
    raw: commandCode,
    type
  };
}

/**
 * Decodes the 10-bit sector code into individual sector numbers.
 * @param {number} sectorCode
 * @returns {number[]}
 */
function griesserSectors (sectorCode) {
  const decoded = decodeGriesserSector(sectorCode);
  if (decoded === 'reserved') {
    return [0];
  }
  if (decoded instanceof Group) {
    const sectors = [];
    for (let i = decoded.start + 1; i <= decoded.end; i++) {
      sectors.push(i);
    }
    return sectors;
  }
  if (typeof decoded === 'number') {
    return [decoded];
  }
  return [0];
}

/**
 * Decodes the German priority description.
 * @param {number} prioCommand priority command / executions
 * @returns {string}
 */
function griesserPrio (prioCommand) {
  switch (prioCommand) {
    case 0: return 'Randbefehl';
    case 1: return 'Automatikbefehl';
    case 3: return 'Prioritätsbefehl';
    case 4: return 'Warnungsbefehl';
    case 5: return 'Sicherheitsbefehl';
    case 6: return 'Gefahrenbefehl';
    default: return 'unbekannte Priorität ' + prioCommand;
  }
}

/**
 * Decodes the priority object containing raw priority value and English priority type description.
 * @param {number} commandCode raw command code (0-63)
 * @param {number} byte2 byte2 value containing priority info
 * @returns {{ raw: number, type: string }|null}
 */
function decodeGriesserPrio (commandCode, byte2) {
  if (commandCode !== 0) return null;
  const prioCommand = byte2 >> 5;
  return {
    raw: prioCommand & 7,
    type: griesserPrio(prioCommand)
  };
}

/**
 * Decodes the parameter/data bytes based on the command.
 * @param {number} command
 * @param {number} byte2
 * @param {number} byte3
 * @param {number} byte4
 * @param {number} byte5
 * @returns {any}
 */
function griesserParameter (command, byte2, byte3, byte4, byte5) {
  let commandOperation;
  switch (command) {
    case 1: // Fahrbefehl
      switch (byte2 & 31) {
        case 0: return { aktion: 'keine Fahrbewegung' };
        case 1: return { aktion: 'obere Endlage' };
        case 2: return { aktion: 'untere Endlage' };
        case 3:
          if ((byte3 >= 1) && (byte3 <= 4)) {
            return { aktion: 'Fixposition Pn anfahren', position: byte3 };
          } else {
            return { aktion: 'unbekannter Wert für Pn', position: byte3 };
          }
        case 4: return { aktion: 'nicht verwendet' };
        case 5: return { aktion: 'Wippen Auf', wippdauer: byte3 };
        case 6: return { aktion: 'Wippen Ab', wippdauer: byte3 };
        case 7: return { aktion: 'Stop' };
        case 8: return { aktion: 'Position anfahren', behanghoehe: byte3, lamellenwinkel: byte4 };
        case 9: return { aktion: 'Behanghöhe anfahren', behanghoehe: byte3 };
        case 10: return { aktion: 'Lamellen stellen', lamellenwinkel: byte3 };
        case 11: return { aktion: 'Szene abrufen', szenenNr: byte3 };
        case 12: {
          const schaltausgangMap = { 0: 'aus', 1: 'ein', 2: 'umschalten' };
          return { aktion: 'Schaltausgang', status: schaltausgangMap[byte3] || ('unbekannt ' + byte3) };
        }
        case 13: return { aktion: 'Parametrierte Sicherheitsposition', nummer: byte3 };
        default:
          return { aktion: 'unbekannter Fahrbefehl', wert: (byte2 & 31) };
      }
    case 2: // Wertkorrektur
      if (byte2 === 0) {
        return { aktion: 'Wertkorrektur', positionAbsolutPercent: byte3 * 5 };
      }
      return { aktion: 'Wertkorrektur', unbekannterTyp: byte2 };
    case 3: // Automatikzustand
      return { modus: byte2, typ: byte3, nummerP3: byte4, nummerP4: byte5 };
    case 4: { // Sperre setzen/löschen
      const fahrbefehlsperre = (byte2 & 1) === 1 ? 'ändern' : 'belassen';
      const tastersperre = (byte2 & 2) === 2 ? 'ändern' : 'belassen';
      const aktion = (byte3 & 1) === 1 ? 'setzen' : 'löschen';

      if (fahrbefehlsperre === 'belassen' && tastersperre === 'belassen') {
        return { aktion: 'keine Sperrenänderung' };
      }

      const targets = [];
      if (fahrbefehlsperre === 'ändern') targets.push('Fahrbefehlsperre');
      if (tastersperre === 'ändern') targets.push('Tastersperre');

      return { aktion: 'Sperre ' + aktion, targets };
    }
    case 5: { // Bedienungs Code
      if (byte2 <= 6) {
        commandOperation = 'Gruppenbedienung';
      } else if (byte2 >= 128 && byte2 <= 134) {
        commandOperation = 'Lokalbedienung';
      } else {
        commandOperation = 'unbekannter Bedienungs Code für ' + byte2;
      }
      const subCommands = [
        'Lang auf',
        'Lang ab',
        'Kurz auf',
        'Kurz ab',
        'Stop',
        'Lang-Kurz auf',
        'Lang-Kurz ab'
      ];
      const subCmdIdx = byte2 & 127;
      const subCmd = subCommands[subCmdIdx] || ('unbekannter Befehl ' + byte2);
      return { bedienung: commandOperation, unterbefehl: subCmd };
    }
    case 6: // Szene setzen
      return { aktion: 'Szene setzen', szenenNr: byte2 };
    case 7: { // Sonderbefehl
      const sonderbefehle = { 0: 'Normalbetrieb + Reset', 1: 'Testmodus', 2: 'Grundeinstellung' };
      return { aktion: 'Sonderbefehl', befehl: sonderbefehle[byte2] || ('unbekannt ' + byte2) };
    }
    case 8: // Datum
      return { jahr: byte2, monat: byte3, tag: byte4 };
    case 9: // Zeit synchronisieren
      if (byte2 === 255) {
        return { aktion: 'synchronisieren' };
      }
      return { stunden: byte2, minuten: byte3 };
    case 10: // Sensorwertmitteilung
      return { kanalnummer: byte2, wertHigh: byte3, wertLow: byte4 };
    case 11: // Busüberwachung
      if (byte2 === 0) return { aktion: 'Überwachung aus' };
      return { aktion: 'Busüberwachung', zykluszeit: byte2 };
    case 16:
    case 19:
    case 22:
    case 24: { // Fahrbereichsgrenzen (Winkel / Höhe)
      const res = { minWinkel: byte2, maxWinkel: byte3 };
      if (command === 24) {
        res.minPosHoehe = byte4;
        res.maxPosHoehe = byte5;
      } else {
        res.minBehanghoehe = byte4;
        res.maxBehanghoehe = byte5;
      }
      return res;
    }
    case 17:
    case 20:
    case 23: // Fahrbereichsgrenzen Position
      return { minPosition: byte2, maxPosition: byte3 };
    default: return { error: 'unbekannter Befehl: ' + command };
  }
}

/**
 * Decode a Griesser object from a hex string.
 *
 * Decoding is based on the official _KNX Non-Standard Datapoint Type Description Form_ filed by Griesser.
 *
 * @param {string} hex
 * @returns {{
 *   sector: number|Group|string,
 *   sectors: number[],
 *   command: { raw: number, type: string },
 *   priority: { raw: number, type: string }|null,
 *   data: any,
 *   raw: string,
 *   sectorCode: number
 * }}
 */
function decodeGriesserObject (hex) {
  if (!hex || typeof hex !== 'string') {
    throw new Error('Invalid hex string');
  }
  const hexBytes = hex
    .replace(/\s+/g, '') // Remove spaces
    .match(/.{1,2}/g); // Chunk into pairs
  if (!hexBytes || hexBytes.length !== 6) {
    throw new Error('Invalid hex string');
  }
  const bytes = hexBytes.map(h => parseInt(h, 16));
  if (bytes.some(isNaN)) {
    throw new Error('Invalid hex string');
  }

  const byte0 = bytes[0];
  const byte1 = bytes[1];
  const byte2 = bytes[2];
  const byte3 = bytes[3];
  const byte4 = bytes[4];
  const byte5 = bytes[5];

  const sectorCode = byte0 + (byte1 & 3) * 256;
  const commandCode = byte1 >> 2;

  return {
    sector: decodeGriesserSector(sectorCode),
    sectors: griesserSectors(sectorCode),
    command: decodeGriesserCommand(commandCode),
    priority: decodeGriesserPrio(commandCode, byte2),
    data: griesserParameter(commandCode, byte2, byte3, byte4, byte5),
    raw: hexBytes.map(h => h.toUpperCase()).join(' '),
    sectorCode
  };
}
